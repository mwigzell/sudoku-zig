## What to build

Replace `Board.grid: grid.Grid` (Box-owned topology per ADR-0005) with flat `[81]Cell` storage on Board. Add accessor methods `cellAt`, `setCell`, `clearCell`. Rename `Cell.locked` → `Cell.given`. Constructors updated to write directly via `cellAt()` + `.given = true` for puzzle givens.

### Why flatten now
Per ADR-0006, the renderer already copies all 81 cells into `[9][9]RenderCell` to draw ASCII lines. Row and column scans under Box ownership require visiting 3 separate Boxes in memory (three cache lines). Flat storage makes row iteration contiguous; column/box are computed views built on simple arithmetic — identical performance but half the indirection.

### 2. Board (`src/board.zig`) — flat storage + accessor methods

Drop the Grid import dependency; Board owns cells directly via row-major indexing.

```zig
const std = @import("std");
const cell = @import("cell.zig");
const renderer = @import("renderer.zig");

pub const DIMENSION_SIZE: u8 = 9;
pub const CELL_COUNT = DIMENSION_SIZE * DIMENSION_SIZE;

pub const Board = struct {
    cells: [CELL_COUNT]Cell,

    pub fn init() Board {
        return .{
            .cells = [_]Cell{ cell.Cell.init(.zero) ** CELL_COUNT },
        };
    }

    /// Return a mutable pointer to the Cell at (row, col).
    /// Indexing is row-major: idx = row * 9 + col.
    pub fn cellAt(self: *Board, row: u4, col: u4) *Cell {
        const idx: usize = @as(usize, @intCast(row)) * DIMENSION_SIZE + @as(usize, @intCast(col));
        return &self.cells[idx];
    }

    /// Set the value of a cell. Does not change its given flag.
    pub fn setCell(self: *Board, row: u4, col: u4, value: CellValue) void {
        const idx: usize = @as(usize, @intCast(row)) * DIMENSION_SIZE + @as(usize, @intCast(col));
        self.cells[idx].value = value;
    }

    /// Clear a cell back to empty and un-mark as given (user can edit again).
    pub fn clearCell(self: *Board, row: u4, col: u4) void {
        const idx: usize = @as(usize, @intCast(row)) * DIMENSION_SIZE + @as(usize, @intCast(col));
        self.cells[idx].value = .zero;
        self.cells[idx].given = false;
    }

    /// Walk the flat cells and produce a render-ready snapshot.
    pub fn assembleRenderSnapshot(self: *Board) renderer.RenderSnapshot {
        var snap: renderer.RenderSnapshot = undefined;
        for (0..DIMENSION_SIZE) |row| {
            for (0..DIMENSION_SIZE) |col| {
                const idx: usize = @as(usize, @intCast(row)) * DIMENSION_SIZE + @as(usize, @intCast(col));
                snap.cells[row][col] = renderer.RenderCell{
                    .value = self.cells[idx].value,
                    .locked = self.cells[idx].given,
                    .conflicting = false, // validator not yet shipped
                };
            }
        }
        return snap;
    }
};
```

### Constructors — `fromFlat` and `fromOneLineString`

Rewrite to use Board's own accessors instead of `grid.cellAt`. The error type stays the same.

```zig
pub fn fromFlat(flat: [CELL_COUNT]u8) BoardError!Board {
    var b = Board.init();

    for (flat, 0..) |v, i| {
        const globalRow: u4 = @intCast(@divTrunc(i, 9));
        const globalCol: u4 = @intCast(@mod(i, 9));
        if (v != 0) {
            const c = b.cellAt(globalRow, globalCol);
            c.value = cell.rawToCellValue(v);
            c.given = true;
        }
    }
    return b;
}

pub fn fromOneLineString(oneLine: []const u8) BoardError!Board {
    if (oneLine.len != CELL_COUNT) return BoardError.WrongLength;

    var flat: [CELL_COUNT]u8 = undefined;
    for (oneLine, 0..) |ch, i| {
        flat[i] = switch (ch) {
            '.' => 0,
            '0' => 0,
            '1'...'9' => ch - '0',
            else => return BoardError.InvalidCharacter,
        };
    }
    return fromFlat(flat);
}

pub const BoardError = error{
    BadCellValue,
    WrongLength,
    InvalidCharacter,
};
```

```zig
test "Board: cellAt resolves correct indices via row-major math" {
    var b = Board.init();

    // Set a known value at (4, 7) → index 4*9 + 7 = 43
    const ptr = b.cellAt(4, 7);
    ptr.value = .seven;

    // Pointer must alias the correct flat cell
    try std.testing.expectEqual(b.cells[4 * DIMENSION_SIZE + 7].value, cell.CellValue.seven);

    // Verify mutation through pointer is visible via direct flat access

    // Edge cases: (0,0) and (8,8)

    try std.testing.expectEqual(cell.CellValue.one, b.cells[0].value);
    b.cellAt(8, 8).value = .nine;
    try std.testing.expectEqual(cell.CellValue.nine, b.cells[80].value);
}
```
**Mechanic:** Prove row-major index math is correct across the range. Mutation through `cellAt` pointer is visible in flat array.
