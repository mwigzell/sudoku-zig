## What to build

BoardView + Row/Col/Box lenses over flat `[81]Cell` storage. Replaces the old Grid→Box indirection with borrowed views and simple index math. Views carry `usize` indices into flat storage; resolution happens through a reference passed at call time — zero-copy iteration over live cells.

### Why now (after bitmask works)

The Board has flat storage (15.1), bitmask cache (15.2). But the only read path is `getCellValue(row, col)` and `assembleRenderSnapshot` (which copies 81 cells). A solver or constraint checker needs to iterate rows, columns, and boxes efficiently — not cell-by-cell. Borrowed views fill that gap without allocating.

### Design notes

- **BoardView** is read-only (`[]const Cell`) — passed from Board via `asView()`. No mutable variant.
- Views store `[9]usize` indices computed at construction time. Resolution requires a `*BoardView` passed to a `.resolve(indices, view)` call — the view struct alone can't resolve values without pointing back at the board.
- All view types are plain data (no methods beyond resolution). Board provides factory functions: `board.asRow(n)`, `board.asCol(n)`, `board.asBox(br, bc)` returning `RowView`, `ColView`, `BoxView`.
- **No Box struct owned on Grid** — the old Grid/Box types become obsolete after this + 15.5 removal.
- Views are intentionally dumb: they carry no state about what they look at beyond their index map. This keeps them cheap to copy, embed inline, and pass between call layers.

## Proposed Code

### 1. View types + resolution — board.zig (new, public side)

```zig
/// Read-only borrowed lens over Board flat storage.
pub const BoardView = struct {
    cells: []const Cell,
    given_bits: u128,

    pub fn get(self: BoardView, row: u4, col: u4) CellValue {
        return self.cells[@as(usize, @intCast(row)) * CELL_COUNT + @as(usize, @intCast(col))].value;
    }

    pub fn isGiven(self: BoardView, row: u4, col: u4) bool {
        const idx: usize = @as(usize, @intCast(row)) * CELL_COUNT + @as(usize, @intCast(col));
        return ((self.given_bits >> @intCast(idx)) & 1) == 1;
    }

    /// Resolve a list of indices against this view — returns [9]CellValue.
    pub fn resolve(self: BoardView, indices: []const usize) [9]CellValue {
        var out: [9]CellValue = undefined;
        for (indices, 0..) |idx, i| {
            out[i] = self.cells[idx].value;
        }
        return out;
    }
};

/// RowView holds flat-storage indices for one horizontal row.
pub const RowView = struct {
    rowNum: u4,
    indices: [9]usize,

    fn getValues(self: RowView, view: BoardView) [9]CellValue {
        return view.resolve(&self.indices);
    }
};

/// ColView holds flat-storage indices for one vertical column.
pub const ColView = struct {
    colNum: u4,
    indices: [9]usize,

    fn getValues(self: ColView, view: BoardView) [9]CellValue {
        return view.resolve(&self.indices);
    }
};

/// BoxView holds flat-storage indices for one 3×3 box.
pub const BoxView = struct {
    boxRow: u2,
    boxCol: u2,
    indices: [9]usize,

    fn getValues(self: BoxView, view: BoardView) [9]CellValue {
        return view.resolve(&self.indices);
    }
};
```

### 2. Board factory methods — asView / asRow / asCol / asBox

Added to `Board` struct:

```zig
    /// Return a borrowed read-only view of this board's cells.
    pub fn asView(self: Board) BoardView {
        return BoardView{ .cells = &self.cells, .given_bits = self.given_bits };
    }

    /// Compute and return RowView indices for global row n (0..8).
    pub fn asRow(self: Board, n: u4) RowView {
        var rv: RowView = undefined;
        rv.rowNum = n;
        const base: usize = @as(usize, @intCast(n)) * CELL_COUNT;
        for (0..CELL_COUNT) |i| rv.indices[i] = base + i;
        return rv;
    }

    /// Compute and return ColView indices for global column n (0..8).
    pub fn asCol(self: Board, n: u4) ColView {
        var cv: ColView = undefined;
        cv.colNum = n;
        const colIdx: usize = @as(usize, @intCast(n));
        for (0..CELL_COUNT) |r| cv.indices[r] = @as(usize, @intCast(r)) * CELL_COUNT + colIdx;
        return cv;
    }

    /// Compute and return BoxView indices for 3×3 box at (boxRow, boxCol).
    pub fn asBox(self: Board, br: u2, bc: u2) BoxView {
        var bv: BoxView = undefined;
        bv.boxRow = br;
        bv.boxCol = bc;

        // base cell index of top-left corner of this box
        const baseR: usize = @as(usize, @intCast(br)) * BOX_DIMENSION;   // actual row start in grid
        const baseC: usize = @as(usize, @intCast(bc)) * BOX_DIMENSION;   // actual col start in grid

        var slot: usize = 0;
        for (0..BOX_DIMENSION) |dr| {
            for (0..BOX_DIMENSION) |dc| {
                const r: usize = baseR + dr;
                const c: usize = baseC + dc;
                bv.indices[slot] = r * CELL_COUNT + c;
                slot += 1;
            }
        }
        return bv;
    }
```

### Why these methods are on Board, not BoardView

Board owns the flat cells slice. View factory needs access to `CELL_COUNT` and `BOX_DIMENSION` constants already in scope. The returned index structs carry no reference back into Board — they're self-contained once indices are baked in. A Board with changed cells produces fresh indices (which don't change) but different values when resolved against the updated view.

### 3. No changes to internal helpers or existing public seam

`getCellValue`, `isGiven`, `setCell`, `clearCell`, `fromFlat`, `fromOneLineString` all unchanged. The new view path is additive only — nothing breaks for consumers using single-cell access.

## Proposed Tests

### Test 1 — asView returns correct cell values at known coordinates

```zig
test "Board: asView resolves same values as getCellValue" {
    const easy_puzzle = blk: {            // re-use puzzle_gen fixture or simple flat...
        var flat: [CELL_COUNT]u8 = undefined;
        @memset(&flat, 0);
        flat[0] = 5;     // A1 = 5
        flat[9 + 4] = 3; // J4 (row 1, col 4) = 3
        break :blk flat;
    };

    var b = try fromFlat(easy_puzzle);
    const view = b.asView();

    try std.testing.expectEqual(CellValue.five, view.get(0, 0));
    try std.testing.expectEqual(CellValue.three, view.get(1, 4));
    try std.testing.expectEqual(CellValue.zero, view.get(8, 8));

    // isGiven consistency: same as Board.isGiven(row, col)
    try std.testing.expect(view.isGiven(0, 0));
    try std.testing.expect(!view.isGiven(1, 5));
}
```

**Mechanic:** Verify the core read seam — BoardView delegates correctly to flat storage. Tests both value and given-status resolution.

### Test 2 — asRow produces contiguous indices and resolves correctly

```zig
test "Board: asRow returns correct view for row 0" {
    var b = Board.init();
    try b.setCell(0, 3, .four);
    try b.setCell(0, 7, .eight);

    const rv = b.asRow(0);
    try std.testing.expectEqual(@as(u4, 0), rv.rowNum);

    // indices must be contiguous: 0..8 (row 0 starts at flat index 0)
    for (0..CELL_COUNT) |i| {
        try std.testing.expectEqual(i, rv.indices[i]);
    }

    const vals = rv.getValues(b.asView());
    try std.testing.expectEqual(CellValue.zero, vals[0]);
    try std.testing.expectEqual(CellValue.four, vals[3]);
    try std.testing.expectEqual(CellValue.eight, vals[7]);
}
```

**Mechanic:** Row indices are contiguous in flat storage (base `row * 9` through `row * 9 + 8`). Verify index layout and value resolution.

### Test 3 — asCol produces strided indices and resolves correctly

```zig
test "Board: asCol returns correct view for col 5" {
    var b = Board.init();
    try b.setCell(2, 5, .six);
    try b.setCell(7, 5, .one);

    const cv = b.asCol(5);
    try std.testing.expectEqual(@as(u4, 5), cv.colNum);

    // indices must stride: 5, 14, 23, ... (col 5 + each row * CELL_COUNT)

        const expected: usize = @as(usize, @intCast(r)) * CELL_COUNT + 5;
        try std.testing.expectEqual(expected, cv.indices[r]);

    const vals = cv.getValues(b.asView());
    try std.testing.expectEqual(CellValue.zero, vals[0]);
    try std.testing.expectEqual(CellValue.six, vals[2]);
    try std.testing.expectEqual(CellValue.one, vals[7]);
    try std.testing.expectEqual(CellValue.zero, vals[8]);
}
```

**Mechanic:** Column indices stride by CELL_COUNT per row. Verify strided layout and resolution across multiple rows.

### Test 4 — asBox produces correct box cell spread

```zig
test "Board: asBox(0, 1) returns top-middle 3×3 box with correct index set" {
    var flat: [CELL_COUNT]u8 = undefined;
    @memset(&flat, 0);
    // Box (0,1) covers rows 0-2, cols 3-5 → flat indices: 3,4,5, 12,13,14, 21,22,23
    flat[4] = 6;   // row 0, col 4 → should appear in box index set (slot 1)
    flat[13] = 2;  // row 1, col 4 → slot 4

    var b = try fromFlat(flat);
    const bv = b.asBox(0, 1);
    try std.testing.expectEqual(@as(u2, 0), bv.boxRow);
    try std.testing.expectEqual(@as(u2, 1), bv.boxCol);

    // Expected indices for box (0,1) — rows 0-2 × cols 3-5 in row-major order
    const expected_indices: [9]usize = .{ 3, 4, 5, 12, 13, 14, 21, 22, 23 };
    for (0..BoxCellCount) |i| {
        try std.testing.expectEqual(expected_indices[i], bv.indices[i]);
    }

    const vals = bv.getValues(b.asView());
    try std.testing.expectEqual(CellValue.six, vals[1]);    // row 0, col 4 → slot 1 
    try std.testing.expectEqual(CellValue.two, vals[4]);   // row 1, col 4 → slot 4
}
```

**Mechanic:** Box indices scatter across non-contiguous rows (3 base rows × 3 base cols). Verify the index map and resolution. This is the hardest view type since indices aren't contiguous or even strided-uniformly like columns.

### Test 5 — BoardView given_bits match live Board state

```zig
test "Board: BoardView.isGiven() tracks Board.given flags through mutation" {
    var flat: [CELL_COUNT]u8 = undefined;
    @memset(&flat, 0);
    flat[19] = 8;   // row 2, col 1 → given

    var b = try fromFlat(flat);
    var view = b.asView();
    
    try std.testing.expect(view.isGiven(2, 1));

    // setCell (on non-given cell) doesn't change its given status
    try b.setCell(5, 6, .three); 
    view = b.asView();
    try std.testing.expect(!view.isGiven(5, 6));

    // clearCell clears the given bit for that cell
    b.clearCell(2, 1);
    view = b.asView();
    try std.testing.expect(!view.isGiven(2, 1));
}
```

**Mechanic:** BoardView borrows `given_bits` from Board — a fresh `asView()` sees mutations. Verifies the borrowed slice pattern works correctly for both value and metadata fields.
